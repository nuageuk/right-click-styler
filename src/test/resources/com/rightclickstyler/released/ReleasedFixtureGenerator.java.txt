package com.rightclickstyler;

import com.google.gson.Gson;
import java.awt.Color;
import java.awt.Graphics2D;
import java.awt.Rectangle;
import java.awt.image.BufferedImage;
import java.io.File;
import java.io.FileOutputStream;
import java.io.OutputStreamWriter;
import java.io.Writer;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Properties;
import org.junit.Test;

/**
 * Throwaway generator run against the RELEASED source (git archive of master). It drives the
 * released ThemeManager exactly as the released panel does and records what it persisted.
 */
public class ReleasedFixtureGenerator
{
	private static final String GROUP = RightClickStylerConfig.GROUP;

	/** One RuneLite profile's rightclickstyler.* values, written as ConfigManager would. */
	private static final class Profile implements RightClickStylerConfig
	{
		final Map<String, String> values = new LinkedHashMap<>();

		@Override
		public String activeTheme()
		{
			return values.getOrDefault(KEY_ACTIVE_THEME, BuiltInThemes.DEFAULT_ID);
		}

		String readThemes()
		{
			return values.get(KEY_USER_THEMES);
		}

		void writeRaw(String key, String value)
		{
			values.put(key, value);
		}

		/** ConfigManager.setConfiguration(group, key, Object) -> objectToString. */
		void setGlobal(String key, Object value)
		{
			String s;
			if (value instanceof Color)
			{
				s = String.valueOf(((Color) value).getRGB());
			}
			else if (value instanceof Enum)
			{
				s = ((Enum<?>) value).name();
			}
			else
			{
				s = value.toString();
			}
			values.put(key, s);
		}
	}

	@Test
	public void generate() throws Exception
	{
		File out = new File(System.getProperty("fixtureOut"));
		out.mkdirs();
		Gson gson = new Gson();
		List<String[]> exports = new ArrayList<>();
		List<String> pixels = new ArrayList<>();

		// ---- Profile "main": a heavily customised user ----
		Profile main = new Profile();
		ThemeManager m = new ThemeManager(main::readThemes, main::writeRaw, main, gson);

		m.applyTheme("midnight");
		m.setWorking(m.effectiveTheme()
			.withBg(new Color(0x101820)).withBorder(new Color(0x334455)).withHeader(new Color(0x0A0A0A))
			.withHeaderText(new Color(0xC0C0C0)).withText(new Color(0xF0F0F0)).withHover(new Color(0x3A5070))
			.withAccent(new Color(0xFFAA00)).withRadius(5).withOpacity(72).withHoverStyle(HoverStyle.BAR)
			.withShowBorder(false).withShowShadow(true).withShowHeader(true).withFont(FontChoice.SANS)
			.withFontSize(13).withTextShadow(false));
		m.saveAsNew("Raid Night");
		String raidNight = m.activeId();

		m.applyTheme("glass");
		m.setWorking(m.effectiveTheme().withHoverStyle(HoverStyle.OUTLINE).withFont(FontChoice.SERIF)
			.withFontSize(11).withShowHeader(false).withOpacity(40));
		m.saveAsNew("Glass Min");

		m.applyTheme("light");
		m.setWorking(m.effectiveTheme().withHoverStyle(HoverStyle.NONE).withFont(FontChoice.RS_SMALL)
			.withRadius(0).withOpacity(100));
		m.saveAsNew("  Solid ✨ Ränge\t ");

		m.applyTheme(raidNight);
		m.duplicate();
		m.rename("Raid Night (alt)");
		m.setWorking(m.effectiveTheme().withOpacity(55).withHoverStyle(HoverStyle.FILL));
		m.save();

		m.applyTheme(raidNight);

		main.setGlobal(RightClickStylerConfig.KEY_HIGHLIGHTS_BOLD, false);
		main.setGlobal(RightClickStylerConfig.KEY_HIGHLIGHT_RULES, "Attack=#FF0000\nPickpocket=#00C8FF\nUse=#A0FF40");
		main.setGlobal(RightClickStylerConfig.KEY_TARGET_MODE, TargetMode.CUSTOM);
		main.setGlobal(RightClickStylerConfig.KEY_NPC_COLOUR, new Color(0x00FF80));
		main.setGlobal(RightClickStylerConfig.KEY_PLAYER_COLOUR, new Color(0xFF66CC));
		main.setGlobal(RightClickStylerConfig.KEY_ITEM_COLOUR, new Color(0x3300FF));
		main.setGlobal(RightClickStylerConfig.KEY_OBJECT_COLOUR, new Color(0xFFD700));
		main.setGlobal(RightClickStylerConfig.KEY_OTHER_COLOUR, new Color(0xAAAAAA));
		main.setGlobal(RightClickStylerConfig.KEY_COMBAT_LEVEL_MODE, CombatLevelMode.CUSTOM);
		main.setGlobal(RightClickStylerConfig.KEY_LOWER_LEVEL_COLOUR, new Color(0x00AA00));
		main.setGlobal(RightClickStylerConfig.KEY_SAME_LEVEL_COLOUR, new Color(0xCCCC00));
		main.setGlobal(RightClickStylerConfig.KEY_HIGHER_LEVEL_COLOUR, new Color(0xDD0000));
		main.setGlobal(RightClickStylerConfig.KEY_TEXT_OFFSET_X, 5);
		main.setGlobal(RightClickStylerConfig.KEY_BASELINE_NUDGE, -1);
		main.setGlobal(RightClickStylerConfig.KEY_SHOW_HIT_ROWS, false);

		// Exports of every saved theme, as the released "Export" action produces them.
		for (Theme t : m.userThemes())
		{
			m.applyTheme(t.id);
			exports.add(new String[]{t.id, t.name, m.exportWorking()});
			pixels.add(t.id + "\t" + samples(t));
		}
		// An export of unsaved edits to a built-in theme.
		m.applyTheme("dark");
		m.setWorking(m.effectiveTheme().withOpacity(63).withAccent(new Color(0x40E0D0)));
		exports.add(new String[]{"(unsaved dark edit)", "Dark", m.exportWorking()});
		pixels.add("(unsaved dark edit)\t" + samples(m.effectiveTheme()));
		for (Theme t : BuiltInThemes.ALL)
		{
			pixels.add(t.id + "\t" + samples(t));
		}
		m.applyTheme(raidNight);

		// ---- Profile "alt": a second RuneLite profile ----
		Profile alt = new Profile();
		ThemeManager a = new ThemeManager(alt::readThemes, alt::writeRaw, alt, gson);
		a.applyTheme("dark");
		a.setWorking(a.effectiveTheme().withOpacity(88).withRadius(2));
		a.saveAsNew("Profile Two Theme");
		a.applyTheme("glass");
		alt.setGlobal(RightClickStylerConfig.KEY_ITEM_COLOUR, new Color(0x00FFFF));
		alt.setGlobal(RightClickStylerConfig.KEY_TEXT_OFFSET_X, 2);

		store(new File(out, "released-profile-main.properties"), main);
		store(new File(out, "released-profile-alt.properties"), alt);
		try (Writer w = new OutputStreamWriter(new FileOutputStream(new File(out, "released-exports.txt")), StandardCharsets.UTF_8))
		{
			w.write("# id<TAB>name<TAB>export, produced by the released ThemeManager.exportWorking()\n");
			for (String[] e : exports)
			{
				w.write(e[0] + "\t" + e[1] + "\t" + e[2] + "\n");
			}
		}
		try (Writer w = new OutputStreamWriter(new FileOutputStream(new File(out, "released-pixels.txt")), StandardCharsets.UTF_8))
		{
			w.write("# theme<TAB>ARGB samples rendered by the released MenuRenderer\n");
			for (String p : pixels)
			{
				w.write(p + "\n");
			}
		}
	}

	private static void store(File file, Profile profile) throws Exception
	{
		Properties p = new Properties();
		for (Map.Entry<String, String> e : profile.values.entrySet())
		{
			p.setProperty(GROUP + "." + e.getKey(), e.getValue());
		}
		try (FileOutputStream os = new FileOutputStream(file))
		{
			p.store(os, "RuneLite profile values written by the released Right Click Styler");
		}
	}

	/** Fill-only samples, away from any text: header, plain row, hovered row (both edges). */
	static String samples(Theme theme)
	{
		BufferedImage image = new BufferedImage(240, 90, BufferedImage.TYPE_INT_ARGB);
		Graphics2D g = image.createGraphics();
		try
		{
			g.setColor(new Color(0x404040));
			g.fillRect(0, 0, 240, 90);
			List<MenuRenderer.Row> rows = Arrays.asList(
				new MenuRenderer.Row("", "", MenuRenderer.TargetKind.OTHER),
				new MenuRenderer.Row("", "", MenuRenderer.TargetKind.OTHER),
				new MenuRenderer.Row("", "", MenuRenderer.TargetKind.OTHER));
			new MenuRenderer().render(g, null, theme, new RightClickStylerConfig() { }, rows,
				new Rectangle(10, 10, 200, 19 + 45), 1, 0);
		}
		finally
		{
			g.dispose();
		}
		int[][] points = {{195, 14}, {195, 36}, {11, 51}, {195, 51}, {195, 66}};
		StringBuilder sb = new StringBuilder();
		for (int[] pt : points)
		{
			if (sb.length() > 0)
			{
				sb.append(',');
			}
			sb.append(pt[0]).append(':').append(pt[1]).append('=')
				.append(Integer.toHexString(image.getRGB(pt[0], pt[1])));
		}
		return sb.toString();
	}
}
